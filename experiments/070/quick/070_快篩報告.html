注意事項

  1. 財報（fin-20261107）、現金流量表與產業別（fincf-20261111）、月營收（revinst-20261111）都是未送驗收的 FinMind 現行修訂值，不是公告當時版本，結果標「時點未驗證」，不得作正式結論。
  2. 報酬大部分來自成交冷清的小型股（流動性組平均報酬砍半），可交易性只做到 L1（一價鎖、停牌、零量），開盤價能否買到實際量未驗證，留待小部位實單驗證。
  3. 面額：共用層面額變更紀錄只有 35 筆、多為 ETF；以 EPS 反推偵測到 2 檔面額不符者已排除，其餘非 10 元面額但未被偵測到者可能高估或低估市值。
  4. 產業別是現行分類回溯；已下市公司多數查無產業別。

結論

建議做正式策略包：10 組全部列入、開樣本外，另補合併組；並建議加做流動性＋持股上限的 E-Test，因為原版雖然強，但持股數與流動性讓它很難實單照做。快篩結果不得作為交易依據。

                             "no_trade", "prev_close", "ref_price", "limit_src", "limit_up", "limit_down", "jump_flag"])

def selection(i, p):
    """換股日 i 收盤後的入選名單（索引陣列）。"""
    rev = M["REV4"][i] if p["rev"] == 4 else M["REV3"][i]

    ok = pool & (mc < p["cap"]) & M["FCF_POS"][i] & M["ROE_POS"][i] & M["OIG_POS"][i]
    pr = np.where(ok, mc / np.where(rev > 0, rev, np.nan), np.inf)
    ok &= pr < p["pr"]
    S = np.where(ok)[0]
    if p.get("top"):

M = dict(np.load("mats070.npz")); meta = pickle.load(open("meta070.pkl", "rb"))
dates, stocks = meta["dates"], meta["stocks"]
T = len(dates)
OA, CA = M["O"] / M["F"], M["C"] / M["F"]
sidx = {s: k for k, s in enumerate(stocks)}
di = {d.date().isoformat(): k for k, d in enumerate(dates)}


def nav(t, combo, top=None, quarterly=False, start="2014-05-16"):
    g = t[t.combo == combo]
    ent, ex = {}, {}
    for r in g.itertuples():
        j = sidx[r.symbol]; a = di[r.entry_date]
        ent.setdefault(a, []).append((r.trade_id, j))
        if isinstance(r.exit_date, str):
            ex.setdefault(di[r.exit_date], []).append(r.trade_id)
    reb = M["q_rebal"] if quarterly else M["rebal_m"]
    reb_exec = set((np.where(reb)[0] + 1).tolist())
    s0 = dates.searchsorted(pd.Timestamp(start))
    pos, cash, navs, turn, nh = {}, 1.0, [], 0.0, []
    for i in range(s0, T):
        for tid, p in pos.items():
            o = OA[i, p[0]]
            if np.isfinite(o) and o > 0 and np.isfinite(p[2]) and p[2] > 0:
                p[1] *= o / p[2]; p[2] = o
        for tid in ex.get(i, []):
            if tid in pos:
                cash += pos[tid][1]; turn += pos[tid][1]; del pos[tid]
        newp = ent.get(i, [])
        for tid, j in newp:
            pos[tid] = [j, 0.0, OA[i, j]]
        total = cash + sum(p[1] for p in pos.values())
        if (i in reb_exec or newp) and pos:
            w = total / max(len(pos), 1)
            if top:
                w = min(w, total / top)
            for p in pos.values():
                turn += abs(w - p[1]); p[1] = w
            cash = total - w * len(pos)
        for tid, p in pos.items():
            c = CA[i, p[0]]
            if np.isfinite(c) and c > 0 and np.isfinite(p[2]) and p[2] > 0:
                p[1] *= c / p[2]; p[2] = c
        navs.append(cash + sum(p[1] for p in pos.values())); nh.append(len(pos))
    s = pd.Series(navs, index=dates[s0:T])
    yrs = (s.index[-1] - s.index[0]).days / 365.25
    return dict(combo=combo, 年化報酬=s.iloc[-1] ** (1 / yrs) - 1, 最大回撤=(s / s.cummax() - 1).min(),
                平均持股數=float(np.mean(nh)), 年換手率=turn / 2 / yrs / float(s.mean()), 期末淨值=s.iloc[-1]), s
